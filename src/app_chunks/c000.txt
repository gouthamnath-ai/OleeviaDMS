// --- FUTURE: Supabase (real backend) — not wired up yet --------------------
// This build runs entirely on window.storage (local, per-browser demo
// storage). Documents (requests, the audit trail, attachment files) are
// written through `storage`; user accounts/passwords always go through
// `localOnly`. Both are plain window.storage right now.
//
// To connect a real Supabase database later, WITHOUT touching any of the
// call sites below (they already use `storage.get/set/delete`):
//   1. npm install @supabase/supabase-js
//   2. Create a Supabase project and run supabase-schema.sql in its SQL
//      editor (creates a "dms_kv" table for documents).
//   3. Replace the body of `storage` below with a version that reads/writes
//      that table via createClient(url, anonKey), falling back to
//      window.storage when Supabase isn't reachable. Leave `localOnly` as
//      window.storage always — user accounts and password hashes should
//      never be sent to Supabase.
//   4. Keep the URL/key out of source control (env vars), and replace the
//      demo "allow all" RLS policy with rules scoped to authenticated users
//      before any real data goes through it.
const storage = {
  get: (key) => window.storage.get(key),
  set: (key, value) => window.storage.set(key, value),
  delete: (key) => window.storage.delete(key),
};
const localOnly = {
  get: (key) => window.storage.get(key),
  set: (key, value) => window.storage.set(key, value),
  delete: (key) => window.storage.delete(key),
};
